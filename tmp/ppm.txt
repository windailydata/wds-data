// ---------- Betting model: our projection + edge on every prop line ----------
	/** Projected stats by normalized player name (from the fantasy projection model) */
	function wds_ppm_proj( $sport ) {
		static $c = array();
		if ( isset( $c[ $sport ] ) ) {
			return $c[ $sport ];
		}
		$m = array();
		if ( function_exists( 'wds_fp_project' ) && function_exists( 'wds_fp_norm' ) && in_array( $sport, array( 'nfl', 'nba', 'mlb', 'nhl' ), true ) ) {
			$P = wds_fp_project( $sport );
			foreach ( (array) ( $P['rows'] ?? array() ) as $r ) {
				$k = wds_fp_norm( $r['name'] );
				if ( isset( $m[ $k ] ) ) {
					continue;
				}
				$s = array();
				foreach ( (array) ( $r['stats'] ?? array() ) as $sk => $sv ) {
					$s[ $sk ] = (float) str_replace( '%', '', (string) $sv );
				}
				$m[ $k ] = array( 's' => $s, 'out' => 'out' === ( $r['ic'] ?? '' ) );
			}
		}
		return $c[ $sport ] = $m;
	}
	/** Projected mean and distribution for one market: array( mean, 'pois'|'norm', sd ) or null */
	function wds_ppm_dist( $sport, $mkey, $s ) {
		$g = function ( $k ) use ( $s ) {
			return isset( $s[ $k ] ) ? (float) $s[ $k ] : null;
		};
		$pois = function ( $m ) {
			return null === $m ? null : array( max( 0.0, $m ), 'pois', 0 );
		};
		$norm = function ( $m, $sd ) {
			return null === $m ? null : array( max( 0.0, $m ), 'norm', max( 0.5, $sd ) );
		};
		switch ( $sport . ':' . $mkey ) {
			case 'nfl:player_pass_yds':
				$m = $g( 'Pass yds' );
				return $norm( $m, max( 30, 0.24 * (float) $m ) );
			case 'nfl:player_pass_tds':
				return $pois( $g( 'Pass TD' ) );
			case 'nfl:player_rush_yds':
				$m = $g( 'Rush yds' );
				return $norm( $m, max( 12, 0.55 * (float) $m ) );
			case 'nfl:player_reception_yds':
				$m = $g( 'Rec yds' );
				return $norm( $m, max( 12, 0.62 * (float) $m ) );
			case 'nfl:player_receptions':
				return $pois( $g( 'Rec' ) );
			case 'nfl:player_anytime_td':
				return $pois( $g( 'Rush/rec TD' ) );
			case 'nba:player_points':
				$m = $g( 'Pts' );
				return $norm( $m, 0.25 * (float) $m + 2 );
			case 'nba:player_rebounds':
				return $pois( $g( 'Reb' ) );
			case 'nba:player_assists':
				return $pois( $g( 'Ast' ) );
			case 'nba:player_threes':
				return $pois( $g( '3PM' ) );
			case 'nba:player_points_rebounds_assists':
				$m = null === $g( 'Pts' ) ? null : $g( 'Pts' ) + (float) $g( 'Reb' ) + (float) $g( 'Ast' );
				return $norm( $m, 0.2 * (float) $m + 3 );
			case 'nhl:player_goal_scorer_anytime':
			case 'nhl:player_goals':
				return $pois( $g( 'Goals' ) );
			case 'nhl:player_points':
				return $pois( null === $g( 'Goals' ) ? null : $g( 'Goals' ) + (float) $g( 'Assists' ) );
			case 'nhl:player_shots_on_goal':
				return $pois( $g( 'Shots' ) );
			case 'nhl:player_assists':
				return $pois( $g( 'Assists' ) );
			case 'mlb:pitcher_strikeouts':
				return $pois( $g( 'K' ) );
			case 'mlb:batter_hits':
				return $pois( $g( 'H' ) );
			case 'mlb:batter_total_bases':
				$m = null === $g( 'H' ) ? null : 1.24 * $g( 'H' ) + 3 * (float) $g( 'HR' );
				return $norm( $m, sqrt( 2.5 * max( 0.2, (float) $m ) ) );
			case 'mlb:batter_home_runs':
				return $pois( $g( 'HR' ) );
			case 'mlb:batter_rbis':
				return $pois( $g( 'RBI' ) );
		}
		return null;
	}
	function wds_ppm_phi( $z ) {
		$t = 1 / ( 1 + 0.2316419 * abs( $z ) );
		$d = 0.3989423 * exp( -$z * $z / 2 );
		$p = $d * $t * ( 0.3193815 + $t * ( -0.3565638 + $t * ( 1.781478 + $t * ( -1.821256 + $t * 1.330274 ) ) ) );
		return $z > 0 ? 1 - $p : $p;
	}
	/** array( P(stat > line), P(stat < line) ) under the projected distribution (pushes excluded) */
	function wds_ppm_ou( $d, $line ) {
		list( $m, $t, $sd ) = $d;
		$line = (float) $line;
		if ( 'pois' === $t ) {
			$k   = (int) floor( $line + 1e-9 );
			$cdf = 0.0;
			$p   = exp( -$m );
			$pk  = 0.0;
			for ( $i = 0; $i <= $k; $i++ ) {
				if ( $i > 0 ) {
					$p *= $m / $i;
				}
				$cdf += $p;
				$pk   = $p;
			}
			$int = abs( $line - round( $line ) ) < 1e-9;
			return array( max( 0, 1 - $cdf ), max( 0, $int ? $cdf - $pk : $cdf ) );
		}
		$z = ( $line - $m ) / $sd;
		return array( 1 - wds_ppm_phi( $z ), wds_ppm_phi( $z ) );
	}
	function wds_ppm_edge_td( $e ) {
		if ( null === $e ) {
			return '<td class="wpp-n wpm-e" data-v="">–</td>';
		}
		$cls = $e >= 0.03 ? 'wpm-pos' : ( $e <= -0.03 ? 'wpm-neg' : 'wpm-mid' );
		return '<td class="wpp-n wpm-e ' . $cls . '" data-v="' . esc_attr( round( $e * 100, 1 ) ) . '">' . ( $e > 0 ? '+' : '' ) . esc_html( number_format( $e * 100, 1 ) ) . '%</td>';
	}

	// Table for one prop market, with our projection and the edge on each side. Sortable; best edge first.
	function wds_pp_table( $sport, $mkey, $list, $limit = 0, $game_col = null ) {
		$info  = wds_pp_markets( $sport )[ $mkey ];
		$model = wds_ppm_proj( $sport );
		$rows  = array();
		foreach ( $list as $player => $p ) {
			$pk = function_exists( 'wds_fp_norm' ) ? wds_fp_norm( $player ) : strtolower( $player );
			$pr = isset( $model[ $pk ] ) && ! $model[ $pk ]['out'] ? wds_ppm_dist( $sport, $mkey, $model[ $pk ]['s'] ) : null;
			$r  = array( 'player' => $player, 'p' => $p, 'd' => $pr, 'eo' => null, 'eu' => null, 'best' => null );
			if ( $info[1] ) {
				if ( $pr && ! empty( $p['yes'] ) ) {
					$r['chance'] = 1 - exp( -$pr[0] );
					$r['eo']     = $r['chance'] - wds_pp_imp( $p['yes']['price'] );
				}
			} elseif ( $pr ) {
				if ( ! empty( $p['over'] ) && null !== $p['over']['point'] ) {
					$r['eo'] = wds_ppm_ou( $pr, $p['over']['point'] )[0] - wds_pp_imp( $p['over']['price'] );
				}
				if ( ! empty( $p['under'] ) && null !== $p['under']['point'] ) {
					$r['eu'] = wds_ppm_ou( $pr, $p['under']['point'] )[1] - wds_pp_imp( $p['under']['price'] );
				}
			}
			$r['best'] = null === $r['eo'] && null === $r['eu'] ? null : max( null === $r['eo'] ? -9 : $r['eo'], null === $r['eu'] ? -9 : $r['eu'] );
			$rows[]    = $r;
		}
		usort( $rows, function ( $a, $b ) {
			if ( null === $a['best'] || null === $b['best'] ) {
				return ( null === $a['best'] ) <=> ( null === $b['best'] );
			}
			return $b['best'] <=> $a['best'];
		} );
		$h  = wds_ppm_assets() . '<div class="wnf-scroll wpp-sortwrap"><table class="wpp-t wpp-sort"><thead><tr><th>Player</th>' . ( $game_col ? '<th>Game</th>' : '' );
		$h .= $info[1] ? '<th>Best odds</th><th data-num="1">Book chance</th><th data-num="1">Our chance</th><th data-num="1">Edge</th><th data-num="1">Books</th>' : '<th data-num="1">Line</th><th data-num="1">Our proj</th><th>Best Over</th><th data-num="1">Over edge</th><th>Best Under</th><th data-num="1">Under edge</th><th data-num="1">Books</th>';
		$h .= '</tr></thead><tbody>';
		foreach ( $rows as $r ) {
			$p  = $r['p'];
			$h .= '<tr><th scope="row">' . esc_html( $r['player'] ) . '</th>' . ( $game_col ? '<td class="wpp-g">' . ( isset( $game_col[ $r['player'] ] ) ? $game_col[ $r['player'] ] : '' ) . '</td>' : '' );
			if ( $info[1] ) {
				$imp = ! empty( $p['yes'] ) ? wds_pp_imp( $p['yes']['price'] ) : null;
				$h  .= '<td>' . wds_pp_cell( $p['yes'], '' ) . '</td><td class="wpp-n" data-v="' . ( null === $imp ? '' : round( $imp * 100, 1 ) ) . '">' . ( null === $imp ? '–' : round( $imp * 100 ) . '%' ) . '</td>';
				$h  .= '<td class="wpp-n wpm-proj" data-v="' . ( isset( $r['chance'] ) ? round( $r['chance'] * 100, 1 ) : '' ) . '">' . ( isset( $r['chance'] ) ? '<b>' . round( $r['chance'] * 100 ) . '%</b>' : '–' ) . '</td>' . wds_ppm_edge_td( $r['eo'] );
			} else {
				$beat = function ( $cell, $over ) use ( $p ) {
					return ( $cell && null !== $p['line'] && null !== $cell['point'] && (float) $cell['point'] !== (float) $p['line'] && ( $over ? (float) $cell['point'] < (float) $p['line'] : (float) $cell['point'] > (float) $p['line'] ) ) ? '<span class="wpp-alt">Better than the ' . esc_html( wds_pp_num( $p['line'] ) ) . ' consensus</span>' : '';
				};
				$pj = $r['d'] ? $r['d'][0] : null;
				$df = ( null !== $pj && null !== $p['line'] ) ? $pj - (float) $p['line'] : null;
				$h .= '<td class="wpp-n" data-v="' . ( null !== $p['line'] ? esc_attr( (float) $p['line'] ) : '' ) . '"><b>' . ( null !== $p['line'] ? esc_html( wds_pp_num( $p['line'] ) ) : '–' ) . '</b></td>';
				$h .= '<td class="wpp-n wpm-proj" data-v="' . ( null === $pj ? '' : esc_attr( round( $pj, 2 ) ) ) . '">' . ( null === $pj ? '–' : '<b>' . esc_html( number_format( $pj, $pj < 10 ? 2 : 1 ) ) . '</b>' . ( null !== $df ? '<small>' . ( $df > 0 ? '+' : '' ) . esc_html( number_format( $df, abs( $df ) < 10 ? 2 : 1 ) ) . ' vs line</small>' : '' ) ) . '</td>';
				$h .= '<td>' . wds_pp_cell( $p['over'], 'O' ) . $beat( $p['over'], true ) . '</td>' . wds_ppm_edge_td( $r['eo'] );
				$h .= '<td>' . wds_pp_cell( $p['under'], 'U' ) . $beat( $p['under'], false ) . '</td>' . wds_ppm_edge_td( $r['eu'] );
			}
			$h .= '<td class="wpp-n" data-v="' . (int) $p['books'] . '">' . (int) $p['books'] . '</td></tr>';
		}
		return $h . '</tbody></table></div>';
	}
